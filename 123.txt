function recommend(){

let math = document.getElementById("math").value;
let coding = document.getElementById("coding").value;
let social = document.getElementById("social").value;

let result = "";

if(math=="yes" && coding=="yes"){
    result="推薦職業：AI工程師";
}
else if(coding=="yes"){
    result="推薦職業：軟體工程師";
}
else if(social=="yes"){
    result="推薦職業：行銷管理";
}
else{
    result="推薦職業：行政工作";
}

document.getElementById("result").innerHTML=result;
}
